import fs from "node:fs"
import { it } from "vitest"

import { GEOGRAPHY_LESSON_PRESET } from "@/features/editor/themes/lib/geography-lesson-preset"
import { presetAsScenes, presetAsTheme } from "@/features/editor/themes/lib/template-catalog"
import { templateSceneDocument } from "@/features/editor/themes/lib/template-preview"

it("dumps the geography lesson scenes", () => {
  const theme = presetAsTheme(GEOGRAPHY_LESSON_PRESET)
  for (const scene of presetAsScenes(GEOGRAPHY_LESSON_PRESET)) {
    const built = templateSceneDocument({ scene, theme, describe: () => undefined })
    if (!built) throw new Error("no document " + scene.name)
    const slug = scene.name.toLowerCase().replace(/[^a-z]+/g, "-")
    fs.writeFileSync(`${process.env.ZZ_GEO_OUT}/tmpl-${slug}.json`, JSON.stringify({ ...built.document, duration: built.seconds }))
  }
})
